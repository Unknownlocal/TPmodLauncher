cmake_minimum_required(VERSION 3.21)
project(modlauncher_core_worker LANGUAGES C)

# Point this at an extracted, verified Frida Core devkit, not a Python wheel.
set(FRIDA_DEVKIT "" CACHE PATH "Directory containing frida-core.h and its static library")
set(FRIDA_SYSTEM_LIBRARIES "" CACHE STRING "System libraries listed by the verified devkit")
set(FRIDA_SYSTEM_FRAMEWORKS "" CACHE STRING "Apple frameworks listed by the verified devkit")
find_library(FRIDA_CORE NAMES frida-core PATHS "${FRIDA_DEVKIT}" NO_DEFAULT_PATH REQUIRED)
add_executable(core-worker src/worker.c)
target_include_directories(core-worker PRIVATE "${FRIDA_DEVKIT}")
target_link_libraries(core-worker PRIVATE "${FRIDA_CORE}")
target_link_libraries(core-worker PRIVATE ${FRIDA_SYSTEM_LIBRARIES})
foreach(framework IN LISTS FRIDA_SYSTEM_FRAMEWORKS)
    find_library(FRIDA_FRAMEWORK_${framework} NAMES "${framework}" REQUIRED)
    target_link_libraries(core-worker PRIVATE "${FRIDA_FRAMEWORK_${framework}}")
endforeach()
set_property(TARGET core-worker PROPERTY C_STANDARD 11)
if(MSVC)
    target_link_libraries(core-worker PRIVATE setupapi)
    # The official Windows devkit uses the static Microsoft C runtime.
    set_property(TARGET core-worker PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    target_compile_options(core-worker PRIVATE /W4 /O2)
    target_link_options(core-worker PRIVATE /OPT:REF /OPT:ICF)
elseif(APPLE)
    target_compile_options(core-worker PRIVATE -Wall -Wextra -Os)
    target_link_options(core-worker PRIVATE -Wl,-dead_strip)
else()
    target_compile_options(core-worker PRIVATE -Wall -Wextra -Os)
    target_link_options(core-worker PRIVATE -Wl,--gc-sections)
endif()
install(TARGETS core-worker RUNTIME DESTINATION .)
